To complete the `calculate` method for a calculator class that evaluates arithmetic expressions, we need to implement three main components:

1. **Tokenization** — Split the input string into numbers and operators.
2. **Infix to Postfix Conversion (Shunting Yard Algorithm)** — Convert the expression from infix to postfix notation to handle operator precedence and parentheses.
3. **Postfix Evaluation** — Use a stack to evaluate the postfix expression.

Here is the complete and well-documented implementation:

---

### ✅ **Complete Calculator Class with `calculate` Method**

```python
class Calculator:
    def __init__(self):
        self.operators = {
            '+': self.add,
            '-': self.subtract,
            '*': self.multiply,
            '/': self.divide
        }

    def add(self, a, b):
        return a + b

    def subtract(self, a, b):
        return a - b

    def multiply(self, a, b):
        return a * b

    def divide(self, a, b):
        if b == 0:
            raise ValueError("Division by zero")
        return a / b

    def tokenize(self, expression):
        """Converts an expression string into a list of tokens (numbers and operators)."""
        tokens = []
        i = 0
        while i < len(expression):
            if expression[i].isspace():
                i += 1
            elif expression[i] in "+-*/()":
                tokens.append(expression[i])
                i += 1
            elif expression[i].isdigit() or expression[i] == '.':
                j = i
                while j < len(expression) and (expression[j].isdigit() or expression[j] == '.'):
                    j += 1
                tokens.append(expression[i:j])
                i = j
            else:
                raise ValueError(f"Invalid character: {expression[i]}")
        return tokens

    def infix_to_postfix(self, tokens):
        """Converts an infix expression to postfix notation using the Shunting Yard algorithm."""
        output = []
        operator_stack = []
        precedence = {'+': 1, '-': 1, '*': 2, '/': 2}

        for token in tokens:
            if token.isdigit() or ('.' in token and token.replace('.', '').isdigit()):
                output.append(token)
            elif token == '(':
                operator_stack.append(token)
            elif token == ')':
                while operator_stack and operator_stack[-1] != '(':
                    output.append(operator_stack.pop())
                if operator_stack and operator_stack[-1] == '(':
                    operator_stack.pop()  # Remove the '('
                else:
                    raise ValueError("Mismatched parentheses")
            else:
                while (operator_stack and
                       operator_stack[-1] != '(' and
                       precedence[operator_stack[-1]] >= precedence[token]):
                    output.append(operator_stack.pop())
                operator_stack.append(token)

        while operator_stack:
            output.append(operator_stack.pop())

        return output

    def calculate(self, expression):
        """
        Evaluates an arithmetic expression.
        Supports +, -, *, /, parentheses, and floating-point numbers.
        """
        tokens = self.tokenize(expression)
        postfix = self.infix_to_postfix(tokens)
        stack = []

        for token in postfix:
            if token.isdigit() or ('.' in token and token.replace('.', '').isdigit()):
                stack.append(float(token))
            else:
                b = stack.pop()
                a = stack.pop()
                result = self.operators[token](a, b)
                stack.append(result)

        if len(stack) != 1:
            raise ValueError("Invalid expression")

        return stack[0]
```

### 🔍 **Usage Example**

```python
calc = Calculator()
print(calc.calculate("3 + 4 * 2"))        # Output: 11.0
print(calc.calculate("1 + 2 * (3 - 4)"))  # Output: -1.0
print(calc.calculate("10 / 2 + 3"))       # Output: 8.0
print(calc.calculate("5 * (2 + 3)"))      # Output: 25.0
```
